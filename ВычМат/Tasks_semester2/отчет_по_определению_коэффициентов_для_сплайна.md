#### отчет по определению коэффициентов для сплайна

даются следующие уравнения для решения системы уравнений со сплайнами
$$
S_i(x)=a_i+b_ih_i+c_ih_i^2+d_ih_i^3\\
h_i = (x-x_i),i = {1,2,...,n-1}\\
\\
S_i(x_{i+1})=S_{i+1}(x_{i+1})=F(x_{i+1})\\
S_i'(x_{i+1})=S_{i+1}'(x_{i+1})\\
S_i''(x_{i+1})=S_{i+1}''(x_{i+1})\\
i={1,2,...,n-2}\\
S_1''(x_1)=0,S_{n-1}''(x_n)=0
\\
$$
допустим n=5
$$
a_1=f_1\\
f_1 + b_1h_1+c_1h_1^2+d_1h_1^3=f_2\\
b_1+2c_1h_1+3d_1h_1^2=b_2\\
2c_1+6d_1h_1=2c_2\\
c_1=0
\\
a_2=f_2\\
f_2 + b_2h_2+c_2h_2^2+d_2h_2^3=f_3\\
b_2+2c_2h_2+3d_2h_2^2=b_3\\
2c_2+6d_2h_2=2c_3\\
\\
a_3=f_3\\
f_3 + b_3h_3+c_3h_3^2+d_3h_3^3=f_4\\
b_3+2c_3h_3+3d_3h_3^2=b_4\\
2c_3+6d_3h_3=2c_4\\
\\
a_4=f_4\\
S_4(x_5)=f_5\\
S_4''(x_5)=0\\
f_4+b_4h_4+c_4h_4^2+d_4h_4^3=f_5\\
2c_4+6d_4h_4=0\\
$$
