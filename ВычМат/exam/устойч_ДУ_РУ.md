$$

$$
\frac{d\vec{x}(t)}{dt}=A\vec{x}(t)+\vec{g}(t)\\
\text{пусть решения системы - }\vec{x}(t),\vec{z}(t)\\
\vec{x}(t)=e^{At}\vec{x}_0 + \int_0^{t}e^{At}\vec{g}(t-\tau)d\tau\\
\vec{z}(t)=e^{At}\vec{z}_0 + \int_0^{t}e^{At}\vec{g}(t-\tau)d\tau\\
\vec{x}(t)-\vec{z}(t)=e^{At}(\vec{x}_0-\vec{z}_0)
